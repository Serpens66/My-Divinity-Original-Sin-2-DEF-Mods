//PROCs
PROC
PROC_CMP_Summoning_Statuses((STRING)_Root, (STRING)_Skill, (GUIDSTRING)_Target, (CHARACTERGUID)_Source)
AND
DB_CMP_SummoningImproved_Statuses(_Root, _Skill, _Status)
THEN
ApplyStatus(_Target, _Status, -1.0, 1, _Source);
PROC_CMP_SetBleedSurface((CHARACTERGUID)_Target, _Status);

//SetSurface
PROC
PROC_CMP_SetBleedSurface((CHARACTERGUID)_Char, (STRING)_Status)
AND
DB_CMP_SummoningImproved_InfusionSurface(_StatusPart, _Surface)
AND
StringContains(_Status, _StatusPart, 1)
THEN
CharacterSetCustomBloodSurface(_Char, _Surface);

//Clear surfaces so slimes wont create a constant chain reaction
PROC
PROC_CMP_Summoning_ClearSurfacesWhenInfusing((GUIDSTRING)_Target)
AND
GetPosition(_Target, _X, _Y, _Z)
THEN
CreateSurfaceAtPosition(_X, _Y, _Z, "SurfaceNone", 1.15, 1.0);


//Wind Up Toy QRY
QRY
QRY_CMP_ToyInfusions((STRING)_Skill, (GUIDSTRING)_Target, (CHARACTERGUID)_Source)
AND
GetTemplate(_Target, _Root)
AND
NOT DB_CMP_SummoningImproved_WindUp_KeepSuicide(_Skill)
AND
_Root == "Summons_WindUpToy_6f8db517-f1af-4b47-b095-f239fd2293d0"
THEN
CharacterRemoveSkill((CHARACTERGUID)_Target, "Shout_SuicideBomberExplosion");
PROC_CMP_Summoning_Statuses(_Root, _Skill, _Target, _Source);

//Event
IF
CharacterStatusApplied(_Target, _Skill, _Source)
AND
CharacterIsSummon(_Target, 1)
AND
GetTemplate(_Target, _Root)
AND
DB_CMP_SummoningImproved_Statuses(_Root, _Skill, _)
AND
NOT QRY_CMP_ToyInfusions(_Skill, _Target, _Source)
THEN
RemoveStatus(_Target, _Skill);
PROC_CMP_Summoning_Statuses(_Root, _Skill, _Target, (CHARACTERGUID)_Source);
PROC_CMP_Summoning_ClearSurfacesWhenInfusing(_Target);

//Reset blood surface for bloated corpse [DOS2EE-8491]
IF
StoryEvent(_Char, "CMP_ResetBloodSurface")
AND
GetTemplate(_Char, _Temp)
AND
_Temp == "Summons_BloatedCorpse_40c6a905-74c3-4d89-9ffe-d3493a22cabd"
THEN
DebugBreak(_Temp);
DebugBreak("Reseting blood surface.");
CharacterSetCustomBloodSurface((CHARACTERGUID)_Char, "SurfaceBlood");
